def fetch_rss(source):
    """抓取 RSS 来源"""
    try:
        feed = feedparser.parse(source["url"])
        articles = []
        for entry in feed.entries[:15]:  # 每源取前15条
            articles.append({
                "title": entry.title,
                "description": entry.get("summary") or entry.get("description", ""),
                "link": entry.link,
                "published": entry.get("published") or entry.get("updated"),
                "source": source["name"],
                "raw_content": entry.get("content", [{}])[0].get("value", "") if entry.get("content") else ""
            })
        print(f"✅ {source['name']} RSS 抓取 {len(articles)} 条")
        return articles
    except Exception as e:
        print(f"❌ {source['name']} RSS 失败: {e}")
        return []

def scrape_page(source):
    """抓取非 RSS 页面（BeautifulSoup）"""
    try:
        resp = requests.get(source["url"], headers=headers, timeout=15)
        resp.raise_for_status()
        soup = BeautifulSoup(resp.text, 'html.parser')
        
        articles = []
        # 根据不同网站调整选择器（示例，实际可微调）
        items = soup.select('article, .announcement-item, .post, .blog-item')[:10]
        for item in items:
            title_tag = item.select_one('h1, h2, h3, .title, a')
            link_tag = item.select_one('a')
            if title_tag and link_tag:
                articles.append({
                    "title": title_tag.get_text(strip=True),
                    "link": link_tag['href'] if link_tag['href'].startswith('http') else "https://www.okx.com" + link_tag['href'],
                    "description": item.get_text(strip=True)[:300],
                    "source": source["name"],
                    "published": datetime.now().isoformat()
                })
        print(f"✅ {source['name']} 页面抓取 {len(articles)} 条")
        return articles
    except Exception as e:
        print(f"❌ {source['name']} 页面失败: {e}")
        return []
