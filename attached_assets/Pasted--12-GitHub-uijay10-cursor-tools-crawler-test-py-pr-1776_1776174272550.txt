# config.py
import asyncio
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception_type
import aiohttp
import sqlite3
import json
import logging
from datetime import datetime

# 你的12类关键词（上面已给出）
from keywords import KEYWORDS

# ============== 配置区 ==============
API_BASE = "https://api.xxx.com/search"   # ← 改成你当前的API地址
HEADERS = {"Authorization": "Bearer YOUR_TOKEN"}  # 如果需要
PROXIES = None  # 需要时填代理池

DB_PATH = "scrape_state.db"
logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")

# ============== 数据库初始化 ==============
def init_db():
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""CREATE TABLE IF NOT EXISTS last_fetch (
        category TEXT PRIMARY KEY,
        last_id TEXT,
        last_time TEXT
    )""")
    conn.commit()
    conn.close()

# ============== 核心抓取函数（带重试+降级） ==============
@retry(stop=stop_after_attempt(5), 
       wait=wait_exponential(multiplier=1, min=2, max=60),
       retry=retry_if_exception_type((aiohttp.ClientError, asyncio.TimeoutError)))
async def fetch_category(session, category: str, query: str):
    # 读取上次抓取位置
    conn = sqlite3.connect(DB_PATH)
    row = conn.execute("SELECT last_id, last_time FROM last_fetch WHERE category=?", (category,)).fetchone()
    since = row[1] if row else "2025-01-01"   # 首次从2025年开始（可改）
    conn.close()

    params = {
        "q": f"{query} since:{since}",
        "limit": 100,
        "lang": "zh,en"   # 支持中英
    }

    async with session.get(API_BASE, params=params, headers=HEADERS, proxy=PROXIES) as resp:
        if resp.status == 429:
            raise aiohttp.ClientError("Rate limit hit")  # 会自动指数退避
        resp.raise_for_status()
        data = await resp.json()

        # 处理结果（你的 prompt.py 可以在这里接）
        processed = []
        for item in data.get("items", []):
            # 用你的 prompt.py 做二次分类/提取
            processed.append(item)

        # 更新状态（增量）
        if processed:
            latest_time = max((p.get("timestamp") for p in processed), default=None)
            if latest_time:
                conn = sqlite3.connect(DB_PATH)
                conn.execute("REPLACE INTO last_fetch VALUES (?, ?, ?)", 
                           (category, processed[-1].get("id"), latest_time))
                conn.commit()
                conn.close()

        return processed

# ============== 主入口（并行抓取） ==============
async def main_scraper():
    init_db()
    async with aiohttp.ClientSession(timeout=aiohttp.ClientTimeout(total=30)) as session:
        tasks = []
        for cat, queries in KEYWORDS.items():
            # 只用最优的那条 query
            query = queries[-1] if len(queries) > 1 else queries[0]
            tasks.append(fetch_category(session, cat, query))
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 统一处理异常
        for cat, res in zip(KEYWORDS.keys(), results):
            if isinstance(res, Exception):
                logging.error(f"[{cat}] 最终失败: {res} → 已自动重试5次")
                # 这里可以加钉钉告警
            else:
                logging.info(f"[{cat}] 成功抓取 {len(res)} 条")
                # 保存到 articles_backup.json 或你的数据库

if __name__ == "__main__":
    asyncio.run(main_scraper())
