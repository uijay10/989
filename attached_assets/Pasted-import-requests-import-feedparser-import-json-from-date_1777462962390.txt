def fetch_rss(source):
    try:
        feed = feedparser.parse(source["url"])
        articles = []
        for entry in feed.entries[:15]:
            articles.append({
                "title": entry.title,
                "description": entry.get("summary") or entry.get("description", ""),
                "link": entry.link,
                "published": entry.get("published") or entry.get("updated"),
                "source": source["name"],
            })
        print(f"✅ {source['name']} → {len(articles)} 条")
        return articles
    except Exception as e:
        print(f"❌ {source['name']} RSS 失败: {e}")
        return []

def scrape_page(source):
    try:
        resp = requests.get(source["url"], headers=headers, timeout=20)
        resp.raise_for_status()
        soup = BeautifulSoup(resp.text, 'html.parser')
        
        articles = []
        selectors = ['article', '.post', '.blog-post', '.announcement', '.news-item', '.entry', '.card', 'h2']
        for sel in selectors:
            items = soup.select(sel)[:12]
            for item in items:
                title_tag = item.select_one('h1, h2, h3, .title, a')
                link_tag = item.select_one('a')
                if title_tag and link_tag and title_tag.get_text(strip=True):
                    link = link_tag.get('href', '')
                    if not link.startswith('http'):
                        link = source["url"].rstrip('/') + '/' + link.lstrip('/')
                    articles.append({
                        "title": title_tag.get_text(strip=True),
                        "link": link,
                        "description": item.get_text(strip=True)[:350],
                        "source": source["name"],
                        "published": datetime.now().isoformat()
                    })
            if articles:
                break
        print(f"✅ {source['name']} 页面 → {len(articles)} 条")
        return articles
    except Exception as e:
        print(f"❌ {source['name']} 页面失败: {e}")
        return []

def fetch_all_sources():
    all_articles = []
    for source in SOURCES:
        if source.get("type") == "rss":
            articles = fetch_rss(source)
        else:
            articles = scrape_page(source)
        all_articles.extend(articles)
        time.sleep(1.5)
    
    # 去重
    unique = {a["title"].strip().lower(): a for a in all_articles if a.get("title")}
    print(f"\n🎉 总共抓取 {len(unique)} 条唯一内容（含第三方聚合器）")
    return list(unique.values())
